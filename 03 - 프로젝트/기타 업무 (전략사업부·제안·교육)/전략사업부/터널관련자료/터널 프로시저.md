### 1. AI 스케일 추가 및 조회.sql

<aside>
💡 ***이 과정은 모든 개발이 끝난 후 현장과 테스트할 때 적용***

</aside>

```sql
-- 스케일추가
INSERT INTO T_AFMF_AI_SCLS01M1 (TUNL_CD, TNRM_INSR_ID, TNRM_TAG_ID, SCAL_USE_YN, SNSR_MSRM_MAX_VAL, SNSR_MSRM_MIN_VAL, PLC_SCAL_MAX_VAL, PLC_SCAL_MIN_VAL, PLC_SCAL_TRSF_VAL, USE_YN, FSTTM_RGSR_ID, FSTTM_RGST_DTTM, LSTTM_MODFR_ID, LSTTM_ALTR_DTTM)
SELECT A.TUNL_CD, A.DEVICE_CODE, A.POINT_CODE, B.SCAL_USE_YN, B.SNSR_MSRM_MAX_VAL, B.SNSR_MSRM_MIN_VAL, B.PLC_SCAL_MAX_VAL, B.PLC_SCAL_MIN_VAL, B.PLC_SCAL_TRSF_VAL, 'Y', 'SYSTEM', SYSDATE, 'SYSTEM', SYSDATE
  FROM TER_MAIN.TB_POINTOPR A
     , (
        SELECT TRIM(TO_CHAR(SUBSTR(TNRM_INSR_ID,1,3),'0000')) MACHINE_CODE
             , TNRM_TAG_ID
             , SCAL_USE_YN
             , SNSR_MSRM_MAX_VAL
             , SNSR_MSRM_MIN_VAL
             , PLC_SCAL_MAX_VAL
             , PLC_SCAL_MIN_VAL
             , PLC_SCAL_TRSF_VAL
          FROM T_AFMF_AI_SCLS01M1
         WHERE 1=1
           AND TUNL_CD = :SELECT_TNCODE
         GROUP BY TRIM(TO_CHAR(SUBSTR(TNRM_INSR_ID,1,3),'0000')), TNRM_TAG_ID, SCAL_USE_YN, SNSR_MSRM_MAX_VAL, SNSR_MSRM_MIN_VAL, PLC_SCAL_MAX_VAL, PLC_SCAL_MIN_VAL, PLC_SCAL_TRSF_VAL
      ) B
 WHERE 1=1
   AND A.MACHINE_CODE = B.MACHINE_CODE
   AND A.POINT_CODE   = B.TNRM_TAG_ID
   AND A.TUNL_CD      = :INPUT_TNCODE
   AND A.MACHINE_CODE = :INPUT_MACHINECODE
 ORDER BY DEVICE_CODE, POINT_CODE
;

-- 풍향풍속계 배율 확인(PLC_SCAL_TRSF_VAL)
select *
from T_AFMF_AI_SCLS01M1
where tunl_cd in (50504)
  and tnrm_insr_id like '202%' 
;

-- 계측기운영 배율 확인(PLC_SCAL_TRSF_VAL)
edit T_AFMF_AI_SCLS01M1
where tunl_cd in (50504)
  and tnrm_insr_id like '808%' 
order by tnrm_insr_id, tnrm_tag_id
;

-- 계측기운영 배율 변경(PLC_SCAL_TRSF_VAL)
update T_AFMF_AI_SCLS01M1
set PLC_SCAL_TRSF_VAL = 100
where tunl_cd in (50504)
  and tnrm_insr_id like '808%'
;
```

UPDATE T_AFMF_SMS_STUP01M1 SET SMS_LTRS_TRNM_YN = 'N', USE_YN = 'N'

                    SELECT INSR_SEQ

```sql
-- 본부, 지사, 터널 정보
SELECT DISTINCT BBCD.INTG_DPTCD BBCODE
    , REPLACE(BBCD.KOR_DPTNM, '본부', '') BBNAME
    , JSCD.INTG_DPTCD JSCODE
    , REPLACE(REPLACE(JSCD.KOR_DPTNM, '지사', ''), '도로관리소', '') JSNAME
    , TCD.TUNL_CD TNCODE
    , REPLACE(TCD.TUNL_NM, '터널', '') TNNAME
 FROM T_AFMF_TNRM_TUNL01M1 TCD
    , (SELECT TB.*, TBC.SORT_TORD FROM V_CNKC_INTG_DPRT01C1 TB, T_AFMF_DPRT_SORT01M1 TBC WHERE TB.INTG_DPTCD = TBC.RSNT_DPTCD) BBCD
    , (SELECT TB.*, TBC.SORT_TORD FROM V_CNKC_INTG_DPRT01C1 TB, T_AFMF_DPRT_SORT01M1 TBC WHERE TB.INTG_DPTCD = TBC.RSNT_DPTCD) JSCD
WHERE 1=1
  AND TCD.HDQR_CD  = BBCD.INTG_DPTCD
  AND TCD.MTNOF_CD = JSCD.INTG_DPTCD
  AND BBCD.INTG_DPTCD   = JSCD.IMDT_HGRN_DPTCD
  AND TNRM_TUNL_TYPE_CD = 'L' -- 터널 타입(L:장터널, M:중터널, S:단터널)  
  AND TUNL_CD IN (60501,70108,70111)
;
select *
from T_AFMF_TNRM_TUNL01M1
```
