1. client에서 회원가입 버튼 클릭 → server로 GET 방식 회원등록요청이 들어간다.

3. 모든 정보 입력후 POST 방식으로 server로 전달 
4. server에 전달된 사용자입력내용 db로 insert
5. 클라이언트에게 보여질 URL을 변경하기 위해 302 : redirect

package member.controller;

import java.io.IOException;
import java.sql.Date;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

import common.HelloMvcUtils;
import member.model.dto.Member;
import member.model.dto.MemberRole;
import member.model.service.MemberService;

/**
 * Servlet implementation class MemberEnrollServlet
 */
@WebServlet("/member/memberEnroll")
public class MemberEnrollServlet extends HttpServlet {
	private static final long serialVersionUID = 1L;
	private MemberService memberService = new MemberService();
	/**
	 * 회원가입폼 요청
	 */
	protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		request
			.getRequestDispatcher("/WEB-INF/views/member/memberEnroll.jsp")
			.forward(request, response);
	}

	/**
	 * 회원정보등록 요청
	 */
	protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		try {
			
			// 1. 인코딩 (filter적용하면 인코딩 생략가능)
			//request.setCharacterEncoding("utf-8");
			
			// 2. 사용자 입력값 처리
			String memberId = request.getParameter("memberId");
			// 단방향 암호화 처리
			String password = HelloMvcUtils.encrypt(request.getParameter("password"), memberId);
			String memberName = request.getParameter("memberName");
			String gender = request.getParameter("gender");
			String email = request.getParameter("email");
			String phone = request.getParameter("phone");
			String address = request.getParameter("address");
			
			String _birthday = request.getParameter("birthday");
			Date birthday = null;
			
			// 사용자가 입력을 하지 않았을 경우 "" 빈문자열 처리가 안됨(변환할수없는타입)
			// 그래서 조건문을 사용해서 처리(null이 올 일은 없지만 안전한 처리를 위해 추가)
			if(_birthday != null && !"".equals(_birthday))
				birthday = Date.valueOf(_birthday);
			
			// db에 저장방식대로 변환해주는 작업. join("구분자",연결할배열)
			// _hobby배열을 콤마로 연결
			String[] _hobby = request.getParameterValues("hobby");
			String hobby = null;
			if(_hobby != null)
				hobby = String.join(",", _hobby);
			//enrollDate는 sysdate 처리되기 때문에 null로 설정함.
			Member member = new Member(
												memberId, password, memberName, MemberRole.U, 
												gender, birthday, email, phone, address, hobby, null);
			System.out.println("member@MemberEnrollServlet = " + member);
			
			// 3. 업무로직 memberService 서비스 요청
			memberService.insertMember(member);
			String msg = "성공적으로 회원가입했습니다.";
			
			// 4. 리다이렉트 (DML처리인 경우 url을 변경해서 새로고침오류를 방지한다.)
			// 사용자 경고창 처리
			// 성공적으로 회원가입했습니다. | 회원가입에 실패했습니다.
			// msg session에 저장해둬야함 request에 저장하면 redirect시 사라짐
			request
				.getSession() 
				.setAttribute("msg", msg);
				response.sendRedirect(request.getContextPath() + "/");
		} catch(Exception e) {
			// 1. 로깅 및 관리팀 알림.
			e.printStackTrace();
			// 2. 예외던지기 - tomcat에 통보(만들어놓은 error.jsp로 넘어갈 수 있게)
			throw e;
		}
	}

}

    	int result = 0;
    	Connection conn = getConnection();
    	try {
    		result = memberDao.insertMember(conn, member);
    		commit(conn); // 정상처리시 commit
    	} catch(Exception e) {
    		rollback(conn); // 오류발생시 rollback

    	} finally {
    		close(conn);
    	}

```java
public int insertMember(Connection conn, Member member) {
		PreparedStatement pstmt = null;
		int result = 0;
		String sql = prop.getProperty("insertMember");
		
		try {
			pstmt = conn.prepareStatement(sql);
			// 미완성쿼리 완성
			pstmt.setString(1, member.getMemberId());
			pstmt.setString(2, member.getPassword());
			pstmt.setString(3, member.getMemberName());
			// MemberRole타입이라(enum) toString으로 U나 A를 넣게함
			pstmt.setString(4, member.getMemberRole().toString()); // "U" "A"
			pstmt.setString(5, member.getGender());
			pstmt.setDate(6, member.getBirthday());
			pstmt.setString(7, member.getEmail());
			pstmt.setString(8, member.getPhone());
			pstmt.setString(9, member.getAddress());
			pstmt.setString(10, member.getHobby());
			
			result = pstmt.executeUpdate();
		} catch(SQLException e) {
			// 예외가 checked예외여서 처리가 복잡 -> runtime예외로 변경 (커스텀예외)
			throw new MemberException("회원가입오류", e);
		} finally {
			close(pstmt);
		}
		return result;
	}

			frm.memberId.value = '<%= memberId %>'; // 현재 입력한 아이디로 부모폼에 입력
			frm.idValid.value = 1; 
			
			self.close(); // 현재창 닫기
		};
		</script>
	<% } else { %>
		<p>[<span id="duplicated"><%= memberId %></span>]는 이미 사용중입니다.</p>
		// 다시 중복검사를 해야하므로 form이 필요함
		<form name="checkIdDuplicateFrm" action="<%= request.getContextPath() %>/member/checkIdDuplicate">
			<input type="text" name="memberId" />
			<input type="submit" value="중복검사" />
		</form>
	<% } %>
	</div>
</body>
</html>
```

### ✔알아야할 내용

#### 1. HTML<form>태그

---

- <form> 태그의 target 속성은 폼 데이터(form data)를 서버로 제출한 후 받는 응답이 열릴 위치를 명시합니다.

속성값

| _blank | 서버로부터 받은 응답을 새로운 윈도우나 탭(tab)에서 보여줌. |
| --- | --- |
| _self | 기본값으로 생략 가능.
서버로부터 받은 응답을 링크가 위치한 현재 프레임에서 보여줌. |
| _parent | 서버로부터 받은 응답을 현재 프레임의 부모 프레임에서 보여줌. |
| _top | 서버로부터 받은 응답을 현재 윈도우 전체에서 보여줌. |
| 프레임 이름 | 서버로부터 받은 응답을 명시된 프레임에서 보여줌. |
