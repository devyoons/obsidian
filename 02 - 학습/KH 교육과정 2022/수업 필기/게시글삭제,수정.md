### Controller : BoardDeleteServlet

---

- 첨부파일 존재시 삭제
    - 첨부파일을 가져오기
        1. `boardService.findByNo(no).getAttachments();` findByNo게시글 조회한 다음에 이 안에서 Attachments() 생성
        2. Service단에 `findAttachmentByBoardNo` 메소드를 생성해도 됨.
    - 파일객체 생성 (실제 파일바라보고있는 파일 객체이다. ⇒ 실제 파일을 제어할 수 있음)
    파일 객체를 쓰임이 다양하다(ex> 디렉토리 바라보기 : 해당파일 리스트할 수 있음)
        - new File(saveDirectory, attach.getRenameFilename());
            - saveDirectory : 부모디렉토리 명시
            - attach.getRenameFilename() : 실제파일명

```java
@WebServlet("/board/boardDelete")
public class BoardDeleteServlet extends HttpServlet {
	private static final long serialVersionUID = 1L;
	private BoardService boardService = new BoardService();

	protected void doPost(HttpServletRequest request, HttpServletResponse response)
			throws ServletException, IOException {

		try {
			// 1.사용자입력처리
			int no = Integer.parseInt(request.getParameter("no"));

			// 2.업무로직
			// 첨부파일 존재시 삭제
			List<Attachment> attachments = boardService.findByNo(no).getAttachments();
			if(attachments != null && !attachments.isEmpty()) 
				for(Attachment attach : attachments) {
					// Attachment에 있는건 renamedFilename 인데 그걸 가지고 File객체생성
					String saveDirectory = getServletContext().getRealPath("/upload/board");
					File delFile = new File(saveDirectory, attach.getRenameFilename());
					if(delFile.exists()) { // 존재하면 삭제
						delFile.delete();
						System.out.println(" > attach.getRenameFilename() =" + attach.getRenameFilename());
					}
				}
			// board 레코드(행) 삭제 (attachment는 on delete cascade에 의해 자동으로 제거된다.)
			int result = boardService.deleteBoard(no);
			
			// 3. redirect : /mvc/board/boardList로 이동
			request.getSession().setAttribute("msg", "게시글을 삭제했습니다.");
			response.sendRedirect(request.getContextPath() + "/board/boardList");

		} catch (Exception e) {
			e.printStackTrace();
			throw e;
		}
	}
}
```

```java
public int deleteBoard(Connection conn, int no) {
		int result = 0;
		PreparedStatement pstmt = null;
		String query = prop.getProperty("deleteBoard"); 
		
		try {
			// delete from board where no = ?
			pstmt = conn.prepareStatement(query);
```

### Controller : BoardUpdateServlet

---

- doGet : 수정폼요청
    1. 사용자 입력값 처리
        1. boardNo 게시판번호
    2. 업무로직
        1. findByNo 만들어놓은 메소드를 통해 게시물번호 가져오기
    3. view단처리
        1. 속성으로 저장후 jsp에 위임
- doPost : DB수정 요청
    1. 제출폼 enctype=”multipart/form-data” 작성
    2. MultipartRequest객체 생성 - 파일저장완료
        1. HttpServletRequest
        2. saveDirectory 
            1. 어플리케이션에서 생명줄이 가장 긴 ServletContext객체 
            ⇒getServletContext().getRealPath(”/upload/board”); 이 경로에 첨부파일저장
        3. maxPostSize 최대업로드크기
        4. encoding
        5. FileRenamePolicy 객체
    3.  사용자입력값처리 - HttpServletRequest가 아닌 MultipartRequest에서 값을 가져오기
        1. 첨부파일 삭제 처리
            1. 서버에 저장된 파일삭제 (서버에 저장된 파일이름은 renamedFilename)
            2. db레코드 제거
    4. 업무로직 - db board, attachment 레코드 등록(insert)
    5. redirect

```java
@WebServlet("/board/boardUpdate")
public class BoardUpdateServlet extends HttpServlet {
	private static final long serialVersionUID = 1L;
	private BoardService boardService = new BoardService();
	/**
	 * 수정폼 요청
	 */
	protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		// 1.사용자입력값 처리
		int no = Integer.parseInt(request.getParameter("no"));
		
		// 2.업무로직
		BoardExt board = boardService.findByNo(no);
		
		// 3.view단처리
		request.setAttribute("board", board);
		request.getRequestDispatcher("/WEB-INF/views/board/boardUpdate.jsp")
			.forward(request, response);
	}

	/**
	 * DB 수정 요청
	 * 
	 * 파일업로드 절차
	 * - 1. 제출폼 enctype="multipart/form-data"
	 * - 2. MultipartRequest객체 생성 - 파일저장완료
	 * 		a. HttpServletRequest
	 * 		b. saveDirectory
	 * 		c. maxPostSize 최대업로드크기
	 * 		d. encoding
	 * 		e. FileRenamePolicy객체 
	 * - 3. 사용자입력값 처리 - HttpServletRequest가 아닌 MultipartRequest에서 값을 가져오기
	 * - 4. 업무로직 - db board, attachment 레코드 등록
	 * - 5. redirect
	 */
	protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		// 2. MultipartRequest객체 생성 - 파일저장완료
		String saveDirectory = getServletContext().getRealPath("/upload/board");
		int maxPostSize = 1024 * 1024 * 10;
		String encoding = "utf-8";
		FileRenamePolicy policy = new HelloMvcFileRenamePolicy();
		MultipartRequest multiReq = 
				new MultipartRequest(request, saveDirectory, maxPostSize, encoding, policy);
		
		// 3. 사용자입력값 처리 - HttpServletRequest가 아닌 MultipartRequest에서 값을 가져오기
		// update board set title = ?, content = ? where no = ?
		int no = Integer.parseInt(multiReq.getParameter("no"));
		String title = multiReq.getParameter("title");
		String memberId = multiReq.getParameter("memberId");
		String content = multiReq.getParameter("content");

		String[] delFiles = multiReq.getParameterValues("delFile");// 삭제하려는 첨부파일pk
		
		// 첨부파일 삭제 처리
		// 1. 서버에 저장된 파일삭제
		// 2. db레코드 제거
		if(delFiles != null) {
			for(String temp : delFiles) {
				int attachNo = Integer.parseInt(temp); // attachment pk
				Attachment attach = boardService.findAttachmentByNo(attachNo);
				// a. 파일 삭제
				File delFile = new File(saveDirectory, attach.getRenameFilename());
				if(delFile.exists()) delFile.delete();
				
				// b. db record 삭제(행)
				int result = boardService.deleteAttachment(attachNo);
				System.out.println("> " + attachNo + "번 첨부파일 삭제!");
			}
		}
		
		BoardExt board = new BoardExt();
		board.setNo(no);
		board.setTitle(title);
		board.setMemberId(memberId);
		board.setContent(content);
		
		File upFile1 = multiReq.getFile("upFile1");
		File upFile2 = multiReq.getFile("upFile2");
		if(upFile1 != null || upFile2 != null) {
			List<Attachment> attachments = new ArrayList<>();
			if(upFile1 != null) // 업로드한 파일명이 
				attachments.add(getAttachment(multiReq, no, "upFile1"));
			if(upFile2 != null)
				attachments.add(getAttachment(multiReq, no, "upFile2"));
			board.setAttachments(attachments);
		}
		
		// 4. 업무로직 
		// - db board(update), attachment(insert) 레코드 등록
		int result = boardService.updateBoard(board);
		
		// 5. redirect
		response.sendRedirect(request.getContextPath() + "/board/boardView?no=" + no);
		
	}

	private Attachment getAttachment(MultipartRequest multiReq, int boardNo, String name) {
		Attachment attach = new Attachment();
		attach.setBoardNo(boardNo);
		attach.setOriginalFilename(multiReq.getOriginalFileName(name));
		attach.setRenameFilename(multiReq.getFilesystemName(name));
		return attach;
	}

}
```
