```jsx
/**
 * prototype을 통한 상속구현
 */
const test1 = () => {
    const books = [
        new Novel('매트릭스', 35000, 0.15, '연애'),
        new Novel('오라클 완전정복', 30000, .2, 'SF'),
        new Novel('네오클래식', 15000, undefined,'심리'),
    ];
    books.forEach((book) => console.log(book));
};

// 부모 생성자 함수
// 매개변수부도 기본값 설정이 가능하다 (discountRate = .05)
function Book(title, price, discountRate = .05){
    this.title = title;
    this.price = price;
    this.discountRate = discountRate;
}

// 할인 적용된 금액 리턴
Book.prototype.getSalePrice = function(){
    return this.price - (this.price * this.discountRate);
}

Book.prototype.toString = function(){
    return `제목: ${this.title}, 판매가: ${this.getSalePrice()}원(정가:${this.price}원)`;
}

// 자식 생성자 함수
// 1. 부모생성자함수 호출 - 부모생성자.apply(자식객체)
// 2. 자식.prototype으로 - 부모.prototype 지정 (Object.create)
// 3. 새로 생성된 부모.prototype객체에 생성자함수 재지정
function Novel(title, price, discountRate, type){
    // 1. 부모생성자함수 호출 - 부모생성자.apply(자식객체)
    Book.apply(this, [title, price, discountRate]);

    this.type = type;
}
// 2. 자식.prototype으로 - 부모.prototype 지정 (Object.create)
// Object.create 전달한 객체를 cloning해서 새로운  객체를 반환
Novel.prototype = Object.create(Book.prototype);

// 3. 새로 생성된 부모.prototype객체에 생성자함수 재지정
Novel.prototype.constructor = Novel; // Book -> Novel
```

### 함수 호출하는 방법 (call, apply, bind)

#### this 객체를 바인딩하면서 함수호출

**`call` : 인자를 낱개로 전달 , this객체를 지정하면서 함수를 호출**

**`apply` : 인자를 배열로 전달**

#### this객체를 바인딩한 함수호출

**`bind` : 호출하지않고 this바인딩만하고 리턴**

![[02 - 학습/KH 교육과정 2022/수업 필기/첨부/602e79b2_Untitled.png]]

this가 현재 window임 전역에 있는 부모가 window라서 

```jsx
const test2 = () => {
    // 이때 화살표함수로 사용하면 this를 처음 지정한 그대로 변경이안됨
    const foo = function(a, b){console.log(this.id, a, b)};
    foo(); // undefined window.id

    // this를 재지정
    const obj = {
        id : 'qwerty1234'
    };
    // 첫번째 인자가 this객체를 지정할 수 있다.(obj를 너의 this로 지정해줌)
    foo.call(obj, 10, 20);
    foo.apply(obj, [10, 20]); // 인자를 배열로 줘야함.

    const bar = foo.bind(obj); // 호출하지 않고, this바인딩후에 리턴
    bar();
};
```


- 생성자 함수의 Syntactic Sugar
- 클래스함수는 생성자를 좀 더 쉽게 사용하기 위한 문법

![[02 - 학습/KH 교육과정 2022/수업 필기/첨부/5b46b0dd_Untitled.png]]

- **`메소드로 작성된 것(sayHi)`**은 **Person.prototype 에 작성**
- **`함수로 작성된 것(callMe)`**은 **현재객체(hong)에 작성됨**
    
    ```jsx
    const test3 = () => {
        const hong = new Person('홍길동', '01012341234');
        console.log(hong);
    
        console.log(typeof Person, Person); // function
    
        hong.sayHi();
        hong.callMe();
    
        // static자원
        console.log(Person.x);
        Person.y();
    };
    
    class Person {
        // Person 생성자 함수 속성으로 작성
        static x = 100;
        static y = function(){
            console.log('yyyyyyyyyyyyyyyyyyyyyyyy');
        };
        // 생성자 함수 : new Person() 호출될 함수
        constructor(name, phone){
            this.name = name;
            this.phone = phone;
        }
        // Person.prototype에 작성 : 메소드단축문법
        sayHi(){
            console.log(`안녕하세요 ${this.name}입니다.`);
        }
    
        // 현재객체에 기록 - 잘사용안함.위 방법으로 사용하는게 메모리적으로 활용적
        callMe = function(){
            console.log(`제 번호는 ${this.phone}입니다. 전화주세요~`);
        };
    }
    ```
    

### 클래스 상속

- `class 자식클래스 extends 부모클래스` : 부모 클래스 상속
- `super()` : 부모 클래스의 생성자 호출. super()를 썼다면 하위 클래스의 초기화 코드는 super() 코드 이후에 만들어져야 한다. 상위 클래스 초기화 이후 하위 클래스 초기화가 가능.

```jsx
class Dev extends Person{
    constructor(name, phone, lang){
        super(name, phone); // 부모생성자 호출
        this.lang = lang;
    }
    // Override
    sayHi(){
        super.sayHi();// 부모 sayHi호출!
        console.log(`저는 ${this.lang}개발자 입니다.`);
    }
    // introduce(){
    //     this.sayHi(); // 만약 오버라이드해서 새로지정했다면 super.sayHi();
    //     console.log(`저는 ${this.lang}개발자 입니다.`);
    // }
}
/**
 * 클래스 상속
 */
const test4 = () => {
    const devs = [
        new Dev('홍길동', '01012341234', 'js'),
        new Dev('신사임당', '01078997899', 'java'),
        new Dev('장영실', '01054545454', 'c++')
    ];
    devs.forEach((dev, index) => {
        // console.log(index, dev);
        dev.sayHi();
    })
};
```
