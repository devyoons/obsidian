```sql
select
    emp_name 직원명,
    job_code 직급코드,
    salary 급여,
    decode(job_code, 'J7', salary * 1.08, 
										 'J6', salary * 1.07,
										 'J5', salary * 1.05, 
										       salary * 1.03) 인상급여
from
    employee;

급여에 반영된 인상률에 대한 결과를 출력해야함.
```
