# 강의: 데이터베이스 기말과제

## 강의 내용

### 인테리어 쇼핑몰 DB 설계

![[02 - 학습/학교/데이터베이스/첨부/644f1b7e_Untitled.png]]

- 회원가입은 소셜로그인도 가능하도록 하기위해 소셜아이디 여부컬럼도 추가. 그러므로 비밀번호는 null일 수도 있도록 하였고, 회원권한으로 관리자인지 사용자인지 판단하기 위한 컬럼도 추가함.
- 배송지에 관한 내용은 따로 테이블을 추가하여 관리하도록 함.
- 상품 테이블과 회원테이블로 장바구니, 주문, 출고 등 관리하도록 하였고 상품은 브랜드와 대분류 소분류로 따로 관리하여 상품이 많아져도 구분하기 쉽도록 함.

```sql
CREATE TABLE `member` (
	`member_id`	varchar2(50)	NOT NULL,
	`password`	varchar2(300)	NULL,
	`member_name`	varchar2(50)	NOT NULL,
	`nickname`	varchar2(100)	NULL,
	`member_role`	char(1)	NOT NULL,
	`phone`	char(11)	NOT NULL,
	`email`	varchar2(200)	NOT NULL,
	`birthday`	date	NULL,
	`gender`	char(1)	NULL,
	`member_social`	varchar2(10)	NOT NULL,
	`enroll_date`	date	NOT NULL
);

CREATE TABLE `address` (
	`address_no`	number	NOT NULL,
	`member_id`	varchar2(50)	NOT NULL,
	`address_title`	varchar2(20)	NOT NULL,
	`post_code`	varchar2(30)	NOT NULL,
	`address`	varchar2(200)	NOT NULL,
	`adress_detail`	varchar2(30)	NOT NULL,
	`address_extra`	varchar2(30)	NULL
);

CREATE TABLE `product` (
	`product_id`	varchar2(80)	NOT NULL,
	`product_name`	varchar2(100)	NULL,
	`main_code`	varchar2(30)	NOT NULL,
	`sub_code`	varchar2(30)	NOT NULL,
	`brand_id`	varchar2(30)	NOT NULL,
	`product_height`	number	NOT NULL,
	`product_width`	number	NULL,
	`product_depth`	number	NOT NULL,
	`product_color`	varchar2(10)	NULL,
	`product_price`	number	NOT NULL,
	`reg_date`	date	NOT NULL,
	`p_content`	varchar2(4000)	NULL
);

CREATE TABLE `brand` (
	`brand_id`	varchar2(30)	NOT NULL,
	`brand_name`	varchar2(30)	NULL
);

CREATE TABLE `main_category` (
	`main_code`	varchar2(30)	NOT NULL,
	`main_category_name`	varchar2(30)	NULL
);

CREATE TABLE `sub_category` (
	`sub_code`	varchar2(30)	NOT NULL,
	`main_code`	varchar2(30)	NOT NULL,
	`sub_category_name`	varchar2(30)	NULL
);

CREATE TABLE `product_io` (
	`no`	number	NOT NULL,
	`product_id`	varchar2(80)	NOT NULL,
	`count`	number	NULL,
	`status`	char(1)	NULL,
	`io_datetime`	timestamp(6)	NULL
);

CREATE TABLE `product_image` (
	`attach_no`	number	NOT NULL,
	`product_id`	varchar2(80)	NOT NULL,
	`original_filename`	varchar2(255)	NULL,
	`renamed_filename`	varchar2(255)	NULL,
	`reg_date`	date	NULL
);

CREATE TABLE `product_desctiption_image` (
	`attach_no`	number	NOT NULL,
	`product_id`	varchar2(80)	NOT NULL,
	`original_filename`	varchar2(255)	NULL,
	`renamed_filename`	varchar2(255)	NULL,
	`red_date`	date	NULL
);

CREATE TABLE `product_stock` (
	`product_id`	varchar2(80)	NOT NULL,
	`stock`	number	NULL
);

CREATE TABLE `cart` (
	`cart_no`	number	NOT NULL,
	`member_id`	varchar2(50)	NOT NULL,
	`product_id`	varchar2(80)	NOT NULL,
	`product_count`	number	NOT NULL
);

CREATE TABLE `orderlist` (
	`order_list_no`	varchar2(255)	NOT NULL,
	`total_price`	number	NULL,
	`order_date`	date	NULL,
	`order_no`	varchar2(255)	NOT NULL
);

CREATE TABLE `purchase` (
	`order_no`	varchar2(255)	NOT NULL,
	`member_id`	varchar2(50)	NOT NULL,
	`product_id`	varchar2(80)	NOT NULL,
	`cart_no`	number	NOT NULL,
	`product_count`	number	NULL,
	`order_date`	date	NULL
);

CREATE TABLE `order` (
	`order_sheet`	varchar2(100)	NOT NULL,
	`order_list_no`	varchar2(255)	NOT NULL
);
```

```sql

ALTER TABLE `purchase` ADD CONSTRAINT `FK_member_TO_purchase_1` FOREIGN KEY (`member_id`)
REFERENCES `member` (`member_id`);
ALTER TABLE `purchase` ADD CONSTRAINT `FK_product_TO_purchase_1` FOREIGN KEY (`product_id`)
REFERENCES `product` (`product_id`);
ALTER TABLE `purchase` ADD CONSTRAINT `FK_cart_TO_purchase_1` FOREIGN KEY (`cart_no`)
REFERENCES `cart` (`cart_no`);
ALTER TABLE `order` ADD CONSTRAINT `FK_orderlist_TO_order_1` FOREIGN KEY (`order_list_no`)
REFERENCES `orderlist` (`order_list_no`);
```
