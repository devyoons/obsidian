# 학습: 사용자 상호작용

## 세부 내용

- alert | console
- confirm
- prompt - template string  ES6

<aside>
🙂 ES6에 추가된 template string 문법은 ```백틱으로 감싸야 한다.

</aside>

```jsx
/**
 * alert 사용자 알림
 * console 객체
 */
function test1(){
  // alert('좋은 아침입니다.');
  
  console.log('좋은 아침입니다.'); // 개발용
  console.error('오류입니다.'); // 오류메세지 출력용
  console.dir({id: 'honggd'}); // 객체를 계층구조로 확인
  
	console.time('abcde'); // 실행시간 체크용
  for(let i = 0; i < 1_000_000; i++){

  }
  console.timeEnd('abcde');
}

/**
 * confirm
 * - 사용자로부터 boolean형 입력을 받을수 있는 메소드
 * - 확인/취소 버튼으로부터 각각 true/false를 반환
 */
function test2(){
  const bool = confirm('정말 삭제하시겠습니까?');
  console.log(typeof bool, bool);

  if(bool){
    // 삭제 프로세스...
    alert('파일이 삭제되었습니다.');
  }
}

/**
 * prompt
 * - 사용자로부터 한줄짜리 입력값을 받는 메소드
 * - 텍스트필드, 확인/취소 버튼을 제어
 * - 확인시 입력값(string)이 리턴, 취소시 null이 리턴
 */
function test3(){
  const name = prompt('이름을 입력하세요');
  console.log(typeof name, name);

  // - 값("", null, undefined)이 없으면 false

  }

}
```
