# 학습: 제어문

## 세부 내용

- 분기문
    - if
        
        ```jsx
        const n = Math.floor(Math.random() * 10 + 1); // 1 ~ 10 사이의 임의의 정수
        console.log(n);
        if(n % 2 == 0){
          console.log("짝수");
        }
        else {
          console.log("홀수");
        }
        ```
        
    - switch
        
        ```jsx
        const n = prompt("1 ~ 3사이의 정수를 입력하세요"); // string을 반환
        switch(n){
          case "1" : console.log('1을 입력하셨습니다.'); break;
          case "2" : console.log('2을 입력하셨습니다.'); break;
          case "3" : console.log('3을 입력하셨습니다.'); break;
          default : console.log('잘못 입력하셨습니다.'); 
        }
        ```
        
- 반복문
    
    <aside>
    💡 반복문안에서 분기처리 break, continue 역시 가능하다.
    
    </aside>
    
    - for
        - `for`
        - `for in` 객체내부 숨겨진 속성`[[Enumerable]]`값이 `true`로 셋팅된 속성(열거형 속성)들만 반복가능.
            - 배열에 대해서 인덱스 참조
        - `for of`  ES6 에 추가된 새로운 <u>컬렉션 전용 반복 구문</u>. for of 구문을 사용하기 위해선 컬렉션 객체가 `[Symbol.iterator]` 속성을 가지고 있어야 함
            - 배열에 대해서 요소 참조
        
        ```jsx
        for(let i = 0; i < 10; i++){
        	console.log(i);
        }
        ```
        
        ```jsx
        // js에는 for..each문은 존재하지않는다.
        const arr = ['a', 'b', 'c'];
        
        // for..in - Enumerable속성이 true인 속성을 순회
        // 배열 사용시 index를 참조
        for(let i in arr){
          console.log(i, arr[i]); // index
        }
        
        // for..of - 해당객체가 iterator속성을 가지고 있는 경우 사용가능
        // 배열 사용시 요소를 참조
        for(let ch of arr){
          console.log(ch);
        }
        ```
        
    - while | do while
        
        ```jsx
        let i = 1;
        while(i <= 10){
          console.log(i++); // 후위증감연산 - 인접연산처리후 증감
        }
        
        let yn;
        do {
          yn = prompt("계속 하시겠습니까? (y/n)", 'y'); // prompt(text, defaultValue)
        } while(yn == 'y');
        ```

    - @실습문제풀이
        
        ```jsx
        /**
         * @실습문제 - 구구단
         * - 사용자 입력한 단수(2 ~ 9)의 구구단을 출력
         * - 2 ~ 9사이의 숫자가 아닌 경우는 출력하지 않는다.
         * - gugudan함수를 호출해서 출력할것
         */
        function test(){
          // number타입으로 명시적 형변환 "2" -> 2
          const dan = Number(prompt("구구단수(2 ~ 9)를 입력하세요", 2));
          if(dan >= 2 && dan <= 9)
            gugudan(dan);
        }
        
        /**
         * 매개변수 dan의 타입은 호출시 매개인자에 따라 결정된다. (Dynamic Typing)
         * 문자형과 숫자형의 산술연산(- * / %)시 자동으로 문자형이 숫자형으로 변환된다.(+ 제외)
         */
        function gugudan(dan){
          console.log(typeof dan, dan); // string 2
          for(let i = 1; i < 10; i++)
            console.log(dan, "*", i, "=", dan * i);
        }
        ```
