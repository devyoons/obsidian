```jsx
/**
 * 객체배열
 */
const test1 = () => {
  const pets = [];
  pets.push({
    name : '구리구리',
    breed : '푸들',
    weight : 3,
    age : 10,
    color : ['white'],
    bark(){
      return this.weight < 10 ? '왈왈' : '멍멍'  
    }
  });
  pets.push({
    name : '둘리',
    breed : '말티즈',
    weight : 6,
    age : 5,
    color : ['white'],
    bark(){
      return this.weight < 10 ? '왈왈' : '멍멍'  
    }
  });
  pets.push({
    name : '사랑이',
    breed : '코카스파니엘',
    weight : 13,
    age : 3,
    color : ['white', 'brown'],
    bark(){
      return this.weight < 10 ? '왈왈' : '멍멍'  
    }
  });

  // console.log(pets);

  // 반복처리
  // - 구리구리야~ 왈왈
  // - 둘리야~ 왈왈
  // - 사랑이야~ 멍멍
  pets.forEach((pet) => {
    // console.log(`${pet.name}야~ ${pet.bark()}`); // 정상출력
    const {name, bark} = pet;
    // method 구조분해할당하면 this를 잃어버린다.
		// console.log(`${name}야~ ${bark()}`); // bark 오작동!
    console.log(`${name}야~ ${bark.call(pet)}`); // this를 pet으로 binding해서 bark 호출!
  });
};
```

```jsx
/**
 * 생성자함수
 * - 관례상 대문자로 시작하는 이름을 갖는다.
 * - new 연산자를 통해 호출되어야 정상적으로 객체를 생성한다.
 * - this용법. 생성자함수안의 this는 현재객체를 가리킨다.
 * - 화살표함수는 생성자함수로 사용할 수 없다.
 */
function Pet(name, breed, weight, age, ...color){
  this.name = name;
  this.breed = breed;
  this.weight = weight;
  this.age = age;
  this.color = color;
	// 매 객체마다 작성되는 방식
  this.bard = function(){
    return this.weight < 10 ? '왈왈' : '멍멍';
  };
}

// 공통메소드를 prototype객체(부모)에 작성하고, 자식객체 공유
Pet.prototype.bark = function(){
	return this.weight < 10 ? '왈왈' : '멍멍';
};

/**
 * 생성자 함수를 통한 객체 생성
 * 
 */
const test2 = () => {
	const pets = [];
	pets.push(new Pet('구리구리', '푸들', 3, 10, 'white'));
	pets.push(new Pet('둘리', '말티즈', 6, 5, 'white'));
	pets.push(new Pet('사랑이', '코카스파니엘', 13, 5, 'white', 'brown'));

	console.log(pets);

	pets.forEach((pet, index) => {
		const info = `반려동물정보
---------------------------------
이름 : ${pet.name}
종 : ${pet.breed}
몸무게 : ${pet.weight}
나이 : ${pet.age}
색깔 : ${pet.color}
짖는 소리 : ${pet.bark()}
---------------------------------`;
		alert(info);
	});
};
```

```jsx
/**
 * @실습문제 
 * 위 csv는 학생별 java, oracle, html, css, js의 점수데이터이다. 
 * 이를 바탕으로 학생객체배열(전역)을 생성하세요. 
 * - 각 객체에 총합을 구하는 속성 getSum메소드
 * - 평균을 구하는 속성 getAvg메소드(소수점첫째자리까지)
 * - 모든 속성정보를 출력하는 toString메소드
 * 
 * 이 객체배열을 콘솔에 toString을 이용해 출력하고,
 * table#students에 표로 화면 출력하세요.
 */
let students = [];

function Student(name, java, oracle, html, css, javascript){
  // 속성 정의
  this.name = name;
  this.java = java;
  this.oracle = oracle;
  this.html = html;
  this.css = css;
  this.javascript = javascript;
  // 메소드 정의
  this.getSum = function(){
      return this.java + this.oracle + this.html + this.css + this.javascript;
  };
  this.getAvg = function(){
      return Math.floor(this.getSum() / 5 * 10) / 10;
  }; 
}

/**
 * str.split(/[/,]/g)
 * - /[/,]/ 정규표현식
 * - []안은 하나의 문자에 대한 옵션 : / 또는 , 
 * - / 또는 ,를 구분자로 해서 split해라.
 * - g 여러번 처리
 */
const test1 = () => {
		students = data.split('\n')
			.map((str) => str.split(/[/,]/g))
      .map((student, index) => {
        return new Student(
            student[0], 
            parseInt(student[1]),
            parseInt(student[2]),
            parseInt(student[3]),
            parseInt(student[4]),
            parseInt(student[5])
        );
    });

    // 출력
    render();
    
};

/**
 * 테이블 화면출력 함수
 */
const render = () => {
    const tbody = document.querySelector("table#students tbody");
    tbody.innerHTML = students.reduce((agg, student, index) => {
        const {name, java, oracle, html, css, javascript} = student;
        return agg   + `
        <tr>
            <td>${index + 1}</td>
            <td>${name}</td>
            <td>${java}</td>
            <td>${oracle}</td>
            <td>${html}</td>
            <td>${css}</td>
            <td>${javascript}</td>
            <td>${student.getSum()}</td>
            <td>${student.getAvg()}</td>
        </tr>`;
    }, "");
};
```

- 생성자함수 파헤치기
    
    자바스크립트 프로토타입 기반 상속모델을 가진 객체지향언어
    
    다음 3가지 구성요소로 이루어진다.
    
    1. 생성자 함수 constructor
    2. prototype객체 (부모객체)
    3. 자식객체 (new 생성자함수() 결과로 만들어진 객체)
    
    <aside>
    🙂 생성된 자식객체는 prototype과 연결된 <u>프로토타입체인</u>을 가지고, `Object.prototype`이 프로토타입 최상위에 위치한다.
    객체의 속성을 검사할때 이 프로토타입체인을 거슬러 올라가며 조회하고, `Object.prototype`까지 검사했을때도 못찾으면 undefined를 리턴한다.
    
    ![[02 - 학습/IT/개발 메모/JavaScript 정리/첨부/javascript prototype.png]]
    
    </aside>
    
```jsx
// 기존 객체생성
// 생성자함수 호출결과로 만들어진 객체와 내용은 동일하지만 Object.prototype을 부모로 한다.
const duck = {
  firstName: "이",
  lastName: "오리",
  say(){
    return "꽥꽥";
  },
};

/**
 * 일반적인 함수이름은 소문자로 시작하는데에 반해 생성자함수는 대문자로 시작하는것이 관례임.
 * Duck 생성자 함수
 */
function Duck(firstName, lastName) {
  this.firstName = firstName;
  this.lastName = lastName;
  this.say = function () {
    return "꽥꽥";
  };
}

/**
* 자바스크립트 프로토타입 기반 상속모델을 가진 객체지향언어
* - 생성된 객체는 prototype과 연결된 프로토타입체인을 가진다.
* - Object.prototype이 프로토타입 최상위에 위치한다.
* 
* - 생성자 함수 Duck 
* - prototype객체(부모객체) 
* - 생성객체 new Duck()
*/
const test3 = () => {
	console.log(Duck); // 생성자함수
  console.log(Duck.prototype); // 프로토타입

  console.log(Duck.prototype.constructor == Duck);

  const lee = new Duck('이', '오리'); // duck객체
  console.log(lee); // Duck {firstName: '이', lastName: '오리'}
  
  lee.say(); // Duck.prototype.say 호출
  console.log(lee.toString()); // [object Object]
  console.log(`${lee}`); // [object Object]

  console.log(lee.__proto__); // Duck.prototype
  console.log(lee.__proto__ === Duck.prototype); // true
  
  // 생성자함수는 new연산자와 함께 호출해야 한다.
  const donald = Duck('Donald', 'Trump');
  console.log(donald); // undefined
  console.log(this.firstName); // Donald
  console.log(this.lastName); // Trump
	
	//(주의) new 키워드 없이 단순호출한 생성자함수는 this->window객체를 가리키고, 속성들은 전역변수가 됨.
	//즉, new키워드로 호출된 생성자함수인 경우만, this가 생성된 객체(prototype object의 속성이 됨. 확인불가)를 가린키게된다.
	const duck2 = Duck("Donald", "Trump");
	console.log(duck2); //undefined 함수 리턴값
	console.log(this.firstName); // Donald
  console.log(this.lastName); // Trump
};
```

- 생성자함수 | 프로토타입객체 | 객체
    - `A` 생성자함수
    - `A.prototype` 프로토타입객체
    - `new A()` 객체 (생성자함수 호출결과)
    
    [‣](https://app.notion.com/p/0f5c10bf5332475794be8f47a29cd9f6) #detail
    
```jsx
function A(){
	this.username = 'Al pacino';
	this.hello = function(){
		console.log('hello');
	}
}

// A의 속성 static
A.username = '홍길동';
A.hello = function(){
	console.log('안녕');
}

// A.prototype의 속성
A.prototype.username = 'sh';
A.prototype.hello = function(){
	console.log('hello sh');
}

/**
 * 속성비교
 * - 생성자함수 속성 static
 * - 프로토타입 속성
 * - 생성객체 속성
 */
const test4 = () => {
  // 객체 (생성자함수 호출결과)
	const x = new A();
	console.log(x.username); // Al pacino
	x.hello(); // hello

	// static
	console.log(A.username); // 홍길동
	A.hello(); // 안녕

	// prototype
	console.log(A.prototype.username); // sh
	console.log(x.__proto__.username); // sh
	
	console.dir(A);
	console.dir(A.prototype);
	console.dir(x);
};
```
